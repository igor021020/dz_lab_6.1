#1. Алгоритм
Начало

Ввод данных: total_seconds — время в секундах (вводит пользователь).

Вычислить значения:

hours = total_seconds / 3600 (целочисленное деление).

minutes = (total_seconds % 3600) / 60.

seconds = total_seconds % 60.

Проверка условия: если hours > 0, то вывести часы с правильным склонением слова «час».

Вывести минуты и секунды с правильным склонением слов «минута» и «секунда» (с помощью вспомогательной функции).

Конец

#2. Реализация программы
```c
#include <stdio.h>
#include <locale.h>

const char* get_form(int n, const char* f1, const char* f2, const char* f5) {
    int n100 = n % 100;
    if (n100 >= 11 && n100 <= 14) return f5;
    int n10 = n % 10;
    if (n10 == 1) return f1;
    if (n10 >= 2 && n10 <= 4) return f2;
    return f5;
}

int main() {
    setlocale(LC_ALL, "Russian");

    int total_seconds;
    int hours, minutes, seconds;

    printf("Введите время в секундах: ");
    if (scanf("%d", &total_seconds) != 1 || total_seconds < 0) {
        printf("Ошибка ввода\n");
        return 1;
    }

    hours = total_seconds / 3600;
    minutes = (total_seconds % 3600) / 60;
    seconds = total_seconds % 60;

    if (hours > 0) {
        printf("%d %s ", hours, get_form(hours, "час", "часа", "часов"));
    }

    printf("%d %s %d %s\n", 
           minutes, get_form(minutes, "минута", "минуты", "минут"),
           seconds, get_form(seconds, "секунда", "секунды", "секунд"));

    return 0;
}
```
